# MDX Blog Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a free, statically-generated MDX blog (index + post pages) to the portfolio, supporting images, embedded links, LaTeX, and syntax-highlighted code blocks.

**Architecture:** Posts are `.mdx` files in `content/blog/`. A small `src/lib/blog.ts` reads/parses them (frontmatter via `gray-matter`). `/blog` lists posts reverse-chronologically; `/blog/[slug]` renders a post with `next-mdx-remote/rsc` plus `remark-math`+`rehype-katex` (LaTeX) and `rehype-pretty-code` (code). Everything renders at build time — no DB, no external service, $0 on the existing Vercel free tier.

**Tech Stack:** Next.js 16 (App Router, RSC), React 19, Tailwind v4, bun, TypeScript (strict), `next-mdx-remote`, `gray-matter`, `remark-math`, `rehype-katex`, `rehype-pretty-code`, `shiki`, `katex`.

## Global Constraints

- Package manager is **bun**. Install with `bun add`; run scripts with `bun run`; run tests with `bun test`.
- Verification commands: `bun run typecheck` (tsc --noEmit), `bun run lint` (eslint), `bun run build` (next build). `bun run check` runs typecheck + lint.
- TypeScript is **strict** — no implicit `any`. Type component props explicitly.
- Next.js 16: route `params` is a **Promise** — always `await params` in page/metadata functions.
- `typedRoutes: true` is enabled — `next/link` `href` to a dynamic path must be cast `as Route` (import `Route` from `next`). Plain `<a href="...">` is NOT type-checked, so it needs no cast.
- Path alias: `@/*` → `./src/*`.
- Theme tokens (from `globals.css`): `--color-primary` `#4ade80`, `--color-primary-hover` `#22c55e`, `--color-terminal-gray` `#0a0a0a`, `--font-display` Orbitron, `--font-mono` JetBrains Mono. Utility classes available: `text-glow`, `terminal-container`, `text-primary`, `font-display`, `font-mono`. Match this dark green-on-black aesthetic.
- Post content images are referenced by public path (e.g. `/cp-website.png` or `/blog/<slug>/x.png`); post-local images live under `public/blog/<slug>/`.

- [ ] **Step 4: Implement the data layer**

Create `src/lib/blog.ts`:
```ts
import fs from "node:fs";
import path from "node:path";
import matter from "gray-matter";

const BLOG_DIR = path.join(process.cwd(), "content", "blog");

export type PostFrontmatter = {
  title: string;
  date: string; // YYYY-MM-DD
  description: string;
  cover?: string;
};

export type PostMeta = PostFrontmatter & { slug: string };
export type Post = PostMeta & { content: string };

export function getAllPosts(): PostMeta[] {
  if (!fs.existsSync(BLOG_DIR)) return [];
  return fs
    .readdirSync(BLOG_DIR)
    .filter((file) => file.endsWith(".mdx"))
    .map((file) => {
      const slug = file.replace(/\.mdx$/, "");
      const raw = fs.readFileSync(path.join(BLOG_DIR, file), "utf8");
      const { data } = matter(raw);
      return { slug, ...(data as PostFrontmatter) };
    })
    .sort((a, b) => (a.date < b.date ? 1 : -1));
}

export function getPost(slug: string): Post | null {
  const filePath = path.join(BLOG_DIR, `${slug}.mdx`);
  if (!fs.existsSync(filePath)) return null;
  const raw = fs.readFileSync(filePath, "utf8");
  const { data, content } = matter(raw);
  return { slug, content, ...(data as PostFrontmatter) };
}
```

- [ ] **Step 2: Write the post page**

Create `src/app/blog/[slug]/page.tsx`:
```tsx
import Link from "next/link";
import { notFound } from "next/navigation";
import type { Metadata } from "next";
import { MDXRemote } from "next-mdx-remote/rsc";
import remarkMath from "remark-math";
import rehypeKatex from "rehype-katex";
import rehypePrettyCode from "rehype-pretty-code";
import "katex/dist/katex.min.css";
import { getAllPosts, getPost } from "@/lib/blog";
import { mdxComponents } from "@/components/mdx";

export function generateStaticParams() {
  return getAllPosts().map((post) => ({ slug: post.slug }));
}

export async function generateMetadata({
  params,
}: {
  params: Promise<{ slug: string }>;
}): Promise<Metadata> {
  const { slug } = await params;
  const post = getPost(slug);
  if (!post) return {};
  return {
    title: post.title,
    description: post.description,
    alternates: { canonical: `/blog/${post.slug}` },
    openGraph: {
      title: post.title,
      description: post.description,
      type: "article",
      ...(post.cover ? { images: [{ url: post.cover }] } : {}),
    },
  };
}

export default async function PostPage({
  params,
}: {
  params: Promise<{ slug: string }>;
}) {
  const { slug } = await params;
  const post = getPost(slug);
  if (!post) notFound();

  return (
    <main className="min-h-screen max-w-3xl mx-auto px-4 pt-24 pb-20">
      <Link href="/blog" className="font-mono text-xs text-gray-500 hover:text-primary">
        ← all posts
      </Link>

      <header className="mt-6 mb-8 border-b border-white/10 pb-6">
        <time className="font-mono text-xs text-gray-500">{post.date}</time>
        <h1 className="font-display text-3xl text-primary text-glow mt-2">{post.title}</h1>
        <p className="text-gray-400 mt-3">{post.description}</p>
      </header>

      <article className="blog-prose">
        <MDXRemote
          source={post.content}
          components={mdxComponents}
          options={{
            mdxOptions: {
              remarkPlugins: [remarkMath],
              rehypePlugins: [
                rehypeKatex,
                [rehypePrettyCode, { theme: "github-dark" }],
              ],
            },
          }}
        />
      </article>
    </main>
  );
}
```

import type { MetadataRoute } from "next";
import { getAllPosts } from "@/lib/blog";

export default function sitemap(): MetadataRoute.Sitemap {
  const baseUrl = process.env.NEXT_PUBLIC_SITE_URL ?? "https://example.com";
  const posts = getAllPosts();

  return [
    {
      url: baseUrl,
      lastModified: new Date(),
      changeFrequency: "monthly",
      priority: 1,
    },
    {
      url: `${baseUrl}/blog`,
      lastModified: new Date(),
      changeFrequency: "weekly",
      priority: 0.8,
    },
    ...posts.map((post) => ({
      url: `${baseUrl}/blog/${post.slug}`,
      lastModified: new Date(post.date),
      changeFrequency: "monthly" as const,
      priority: 0.6,
    })),
  ];
}
